% ===========================================================================
%  PART 2 --- Rotations and Insertion
%  Flow: the string analogy (frame by frame) -> what actually moves ->
%        the four cases -> insertion rule -> insertion simulation
%
%  The string analogy is drawn EXACTLY:
%    x of every node = its position in sorted order  (FROZEN, never changes)
%    y of every node = its depth in the current tree
%  A rotation therefore changes only heights. The string cannot reorder.
% ===========================================================================

  bead 5 before touching it. Pull: the curved arrow shows the swing direction
  as everything else hangs down from the new pivot. The faded beads are where
  3 and 8 used to sit --- watch them settle into their new spot. Pause again:
  let go, point at bead 3. Pull: same swing, new pivot, and now 5 and 8 are the
  ones fading into their new position. Look at the bottom line the whole time.
  The order never moves. Only the grip changes. That is a rotation.}
\end{frame}

\begin{frame}[label=p2]{Four cases, and what to do}

  \begin{tabular}{l|l|l}

        \draw[avledge] (r\i) -- (m\i);
        \draw[avledge] (m\i) -- (l\i);

    }
  \end{tikzpicture}
  \end{center}

  \vspace{0.2em}
  \begin{center}
    {\color{avlmuted}\small A double rotation is not new ---
     it is just two single rotations, one after the other.}
  \end{center}

  \note{Four shapes, four fixes. The two straight ones need a single rotation.
  The two bent ones need two rotations: first straighten the bend, then rotate
  as normal.}
\end{frame}

\begin{frame}[label=p2]{Insertion: the rule}
  \vspace{0.9em}
  \begin{center}
  \begin{tikzpicture}[x=1cm,y=1cm]
    \node[draw=avlptwo,fill=avlptwo!8!avlground,rounded corners=3pt,
          minimum width=3.4cm,minimum height=1.1cm,font=\small,text=avlink] (s1) at (-4.4,0)
          {\begin{minipage}{3.0cm}\centering Insert like a normal BST\end{minipage}};
    \node[draw=avlptwo,fill=avlptwo!8!avlground,rounded corners=3pt,
          minimum width=3.4cm,minimum height=1.1cm,font=\small,text=avlink] (s2) at (0,0)
          {\begin{minipage}{3.0cm}\centering Walk back up to the root\end{minipage}};
    \node[draw=avlfib,fill=avlfib!10!avlground,rounded corners=3pt,
          minimum width=3.4cm,minimum height=1.1cm,font=\small,text=avlink] (s3) at (4.4,0)
          {\begin{minipage}{3.0cm}\centering Rotate at the \textbf{first} bad node\end{minipage}};
    \draw[avlgoldarrow,line width=1.1pt] (s1) -- (s2);
    \draw[avlgoldarrow,line width=1.1pt] (s2) -- (s3);
  \end{tikzpicture}
  \end{center}

  \vspace{1.6em}
  \begin{center}
    {\large\color{avlfib}\bfseries One rotation is always enough.}\\[0.7em]
    {\color{avlmuted}After it, the subtree is back to the height it had
     \emph{before} the insert.}\\[0.3em]
    {\color{avlmuted}So no node above it sees any change, and the repair stops.}
  \end{center}

  \note{Three steps. Insert normally, walk up, fix the first bad node you meet.
  And one rotation always finishes the job, because it puts the subtree back to
  its original height, so nothing above notices.}
\end{frame}
