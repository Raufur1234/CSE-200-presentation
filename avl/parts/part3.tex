%  PART 3 --- Deletion, the worst case simulation, and the Fibonacci payoff
%  Flow: deletion rule -> why deletion is worse -> every layer is a Fibonacci
%        tree -> worst case simulation run on that fact -> the Fibonacci match
%
%  Every number is simulated ground truth:
%    worst-case deletion rotations = floor(h/2) exactly, h = 2..14
%    worst-case insertion rotations = 1 at every height tested
%    h=7 tree: deleting the leaf of in-order rank 33 (pre-order f34) fires
%      3 rotations, at ranks 32, 29, 21 (pre f31, f23, f2), depths 3 -> 2 -> 1
%    per-layer subtree sizes: 54 33 20 12 7 4 2 1, and the rotation sites are
%      the T_2, T_4, T_6 layers -- every second one

%  all of the above before it writes tikz/cascade.tex. Re-run it after any
%  change to the tree, and never hand-edit the generated file.
% ===========================================================================

% ------------------------------------------------------------------ why worse
\begin{frame}[label=p3]{Why deletion cannot stop}
  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \vspace{0.2em}
      {\bfseries\color{avlfib}Insertion}\\[0.4em]
      {\color{avlmuted}The rotation puts the subtree back to its
       \emph{old} height.}\\[0.4em]
      The parent sees no change.\\[0.3em]
      \fib{So the repair stops.}\\[1.1em]

      {\bfseries\color{avlalert}Deletion}\\[0.4em]
      {\color{avlmuted}The rotation leaves the subtree \emph{one shorter}.}\\[0.4em]
      That is itself a change.\\[0.3em]
      \viol{So the parent may break too.}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1cm,y=1cm,scale=0.92]
        \node[avltri,draw=avlfib,minimum size=2.0cm] at (-1.4,0) {};
        \node[color=avlfib,font=\scriptsize] at (-1.4,-1.45) {height $h$};
        \node[color=avlmuted,font=\scriptsize] at (-1.4,1.20) {after insert};

        \node[avltri,draw=avlalert,minimum size=1.65cm] at (1.5,-0.18) {};
        \node[color=avlalert,font=\scriptsize] at (1.5,-1.45) {height $h-1$};
        \node[color=avlmuted,font=\scriptsize] at (1.5,1.20) {after delete};

        \draw[draw=avlalert,line width=1.2pt,-{Stealth[length=4pt]}]
          (1.5,-1.85) -- (1.5,-2.45);
        \node[color=avlalert,font=\scriptsize,anchor=north] at (1.5,-2.55)
          {parent must be checked};
        \draw[draw=avlalert,line width=1.2pt,-{Stealth[length=4pt]}]
          (1.5,-3.05) -- (1.5,-3.65);
        \node[color=avlalert,font=\scriptsize,anchor=north] at (1.5,-3.75)
          {and its parent \dots};
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

  \note{This is the whole asymmetry in one picture. Insertion's rotation
  restores the height, so the parent never finds out. Deletion's rotation
  shortens the subtree, and a shorter subtree is exactly what the parent was
  depending on not happening.}
\end{frame}

  that makes the next slide behave the way it does: nothing here has slack.}

% ------------------------------------------------------------ WORST CASE SIM
%  The cascade, run for real. Every overlay below is a DIFFERENT tree: the node
%  is actually removed, and each rotation actually restructures the picture.
%  All five states come from verify/cascade_gen.py driving a real AVL
%  implementation; the geometry is read in from tikz/cascade.tex.
%
%  Layout rule (the same one Part 2 states): a node's x is its ORIGINAL
%  in-order rank and never changes. Only depth changes. So a rotation moves
%  nodes straight up or down, the deleted node leaves a permanent gap at its
%  own x, and the height drop at the end is a visible change in the picture
%  rather than a claim in the caption.
\newcommand{\cascdx}{0.196}
\newcommand{\cascdy}{0.50}
\newcommand{\cascrad}{1.5pt}

% \casctree{<nodes>}{<edges>} -- edges behind, discs on top
\newcommand{\casctree}[2]{%
  \foreach \r/\d in #1 {\coordinate (c-\r) at (\r*\cascdx,-\d*\cascdy);}%
  \foreach \a/\b in #2 {\draw[avlgoldedge] (c-\a) -- (c-\b);}%
  \foreach \r/\d in #1 {\fill[avlfib] (c-\r) circle (\cascrad);}}

% the node that is gone, marked at the x-slot it will never occupy again
\newcommand{\cascgap}{%
  \draw[draw=avlalert!55, dashed, line width=0.7pt]
    (\casctarget*\cascdx,-4*\cascdy) circle (4.2pt);}

% \cascbreak{<rank>}{<bf>} -- the node that is now illegal
\newcommand{\cascbreak}[2]{%
  \draw[draw=avlalert, line width=1.2pt] (c-#1) circle (4.2pt);
  \node[text=avlalert, font=\scriptsize\bfseries, anchor=west,
        fill=avlground, inner sep=1.6pt, rounded corners=1pt,
        xshift=5.5pt, yshift=7pt] at (c-#1) {$\mathrm{bf}=#2$};}

% height ruler down the left edge; #1 = height, #2 = colour
\newcommand{\cascruler}[2]{%
  \begin{scope}[xshift=-0.55cm]
    \draw[decorate,decoration={brace,amplitude=4pt},#2,line width=0.8pt]
      (0,0.10) -- (0,{-#1*\cascdy-0.10});
    \node[color=#2,font=\scriptsize,anchor=east,rotate=90]
      at (-0.26,{-#1*\cascdy/2}) {height #1};
  \end{scope}}

\begin{frame}[label=p3]{Worst case: one leaf out, and it does not stop}
  \vspace{-0.5em}
  \begin{center}
  \begin{tikzpicture}[x=1cm,y=1cm]
    % fixed extent: the picture must not resize between clicks
    \path (-1.5,-3.95) rectangle (11.2,0.60);

    \only<1>{\casctree{\cascNZero}{\cascEZero}%
             \draw[draw=avlalert,line width=1.2pt]
               (\casctarget*\cascdx,-4*\cascdy) circle (3.6pt);
             \cascruler{\cascHZero}{avlfib}}
    \only<2>{\casctree{\cascNOne}{\cascEOne}\cascgap
             \cascbreak{\cascHotOne}{\cascBfOne}\cascruler{\cascHOne}{avlfib}}
    \only<3>{\casctree{\cascNTwo}{\cascETwo}\cascgap
             \cascbreak{\cascHotTwo}{\cascBfTwo}\cascruler{\cascHTwo}{avlfib}}
    \only<4>{\casctree{\cascNThree}{\cascEThree}\cascgap
             \cascbreak{\cascHotThree}{\cascBfThree}\cascruler{\cascHThree}{avlfib}}
    \only<5->{\casctree{\cascNFour}{\cascEFour}\cascgap
             \cascruler{\cascHFour}{avlpthree}}
  \end{tikzpicture}
  \end{center}

  \vspace{-0.7em}
  \begin{center}
    \only<1>{{\color{avlmuted}54 nodes, height 7, no spare room anywhere.
              Delete the \viol{marked leaf}.}}
    \only<2>{{\color{avlmuted}Gone. Its grandparent is now
              \viol{$\mathrm{bf}=+2$} --- illegal. \fib{Rotate.}}}
    \only<3>{{\color{avlmuted}Fixed --- but that subtree got \emph{shorter},
              so its own parent is now \viol{$+2$}. \fib{Rotate again.}}}
    \only<4>{{\color{avlmuted}And again. This one is a child of the root.
              \fib{Rotate a third time.}}}
    \only<5->{{\large\viol{\textbf{3 rotations}}\quad\quietc{$\cdot$}\quad
               54 $\rightarrow$ 53 nodes \quad\quietc{$\cdot$}\quad
               height \textbf{7} $\rightarrow$
               \color{avlpthree}\textbf{6}}\\[0.3em]
              {\color{avlmuted}The height only fell when the cascade reached
               the top --- and one delete paid for all of it.}}
  \end{center}

  \note{One click at a time, and say what changed. The leaf is gone --- the
  dashed circle is the hole it left, and it stays there all the way through.
  Now its grandparent has balance factor plus two, so it must rotate. Click.
  That rotation fixed the node but made the subtree one shorter, and shorter is
  exactly what its parent was relying on not happening --- plus two again.
  Click. A third time, now a child of the root. Click. Watch the ruler on the
  left: the height only drops at the very end. Three rotations, for one leaf,
  because a minimal tree has no slack to absorb anything.}
\end{frame}
