    \node[draw=avlalert,fill=avlalert!8!avlground,rounded corners=3pt,
          minimum width=3.6cm,minimum height=1.15cm,font=\small,text=avlink] (s3) at (4.5,0)

    \draw[avlarrow,line width=1.1pt] (s2) -- (s3);
  \end{tikzpicture}
  \end{center}

  \vspace{1.5em}
  \begin{center}
    {\large\color{avlink}The first two steps are the same as insertion.}\\[0.8em]
    {\large\viol{\bfseries The third one is not.}}\\[0.7em]
    {\color{avlmuted}With insertion you stop after one rotation.
     With deletion you \emph{cannot} stop.}
  \end{center}

  \note{The first two steps look identical to insertion. The difference is the
  last word. Insertion fixes one node and stops. Deletion may have to fix every
  node on the way up.}
\end{frame}

% ------------------------------------------------------------------ why worse
\begin{frame}[label=p3]{Why deletion cannot stop}
  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \vspace{1.0em}
      {\bfseries\color{avlfib}Insertion}\\[0.4em]
      {\color{avlmuted}The rotation puts the subtree back to its
       \emph{old} height.}\\[0.4em]
      The parent sees no change.\\[0.3em]
      \fib{So the repair stops.}\\[1.6em]

      {\bfseries\color{avlalert}Deletion}\\[0.4em]
      {\color{avlmuted}The rotation leaves the subtree \emph{one shorter}.}\\[0.4em]
      That is itself a change.\\[0.3em]
      \viol{So the parent may break too.}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1cm,y=1cm,scale=0.92]
        \node[avltri,draw=avlfib,minimum size=2.0cm] at (-1.4,0) {};
        \node[color=avlfib,font=\scriptsize] at (-1.4,-1.45) {height $h$};
        \node[color=avlmuted,font=\scriptsize] at (-1.4,0.95) {after insert};

        \node[avltri,draw=avlalert,minimum size=1.65cm] at (1.5,-0.18) {};
        \node[color=avlalert,font=\scriptsize] at (1.5,-1.45) {height $h-1$};
        \node[color=avlmuted,font=\scriptsize] at (1.5,0.95) {after delete};

        \draw[draw=avlalert,line width=1.2pt,-{Stealth[length=4pt]}]
          (1.5,-1.85) -- (1.5,-2.6);
        \node[color=avlalert,font=\scriptsize,anchor=north] at (1.5,-2.7)
          {parent must be checked};
        \draw[draw=avlalert,line width=1.2pt,-{Stealth[length=4pt]}]
          (1.5,-3.25) -- (1.5,-3.9);
        \node[color=avlalert,font=\scriptsize,anchor=north] at (1.5,-4.0)
          {and its parent \dots};
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

  \note{This is the whole asymmetry in one picture. Insertion's rotation
  restores the height, so the parent never finds out. Deletion's rotation
  shortens the subtree, and a shorter subtree is exactly what the parent was
  depending on not happening.}
\end{frame}

% ------------------------------------------------------------------ WORST CASE SIM
\begin{frame}[label=p3]{Worst case simulation: delete one leaf}
  \vspace{-0.4em}
  \begin{center}
  \begin{tikzpicture}[x=1cm,y=1cm,
      avldot/.style={avlgolddot},
      avledge/.style={avlgoldedge}]
    \setavlscale{0.55}{0.58}
    \resetavl
    \drawfib{7}{0}{0}

    % Node ids are NOT guessed -- verify/locate_nodes.ps1 maps the verified
    % worst-case target (in-order key 33) and its rotation-site ancestors to
    % the pre-order ids this macro emits:
    %   target -> f34 (depth 4); sites depth 3 -> f31, 2 -> f23, 1 -> f2
    %
    % Each rotation gets TWO beats: a PAUSE (circle the node, nothing moves
    % yet) then the ARC (the rotation fires). That is the "cover, then
    % animate" cadence -- the audience sees the trigger before the fix.
    \only<2->{\draw[draw=avlalert,line width=1.1pt] (f34) circle (0.20);}

    % --- rotation 1 : pause at f31, then fire ---
    \only<3>{\draw[draw=avlalert,line width=1.3pt] (f34) -- (f31);
             \draw[draw=avlalert,line width=1.4pt] (f31) circle (0.30);}
    \only<4->{\draw[draw=avlalert,line width=1.3pt] (f34) -- (f31);}
    \only<4>{\rotarcalert{f31}{0.42}{200}{340}
             \fill[avlalert] (f31) circle (1.6pt);}
    \only<5->{\draw[draw=avlalert,line width=1.1pt] (f31) circle (0.20);}

    % --- rotation 2 : pause at f23, then fire ---
    \only<5>{\draw[draw=avlalert,line width=1.3pt] (f31) -- (f23);
             \draw[draw=avlalert,line width=1.4pt] (f23) circle (0.30);}
    \only<6->{\draw[draw=avlalert,line width=1.3pt] (f31) -- (f23);}
    \only<6>{\rotarcalert{f23}{0.42}{200}{340}
             \fill[avlalert] (f23) circle (1.6pt);}
    \only<7->{\draw[draw=avlalert,line width=1.1pt] (f23) circle (0.20);}

    % --- rotation 3 : pause at f2, then fire ---
    \only<7>{\draw[draw=avlalert,line width=1.3pt] (f23) -- (f2);
             \draw[draw=avlalert,line width=1.4pt] (f2) circle (0.30);}
    \only<8->{\draw[draw=avlalert,line width=1.3pt] (f23) -- (f2);}
    \only<8>{\rotarcalert{f2}{0.42}{200}{340}
             \fill[avlalert] (f2) circle (1.6pt);}
    \only<9->{\draw[draw=avlalert,line width=1.3pt] (f2) -- (f1);}

    % --- height ruler on the left: visibly drops 7 -> 6 ---
    \begin{scope}[xshift=-6.6cm]
      \only<1-8>{
        \draw[decorate,decoration={brace,amplitude=5pt},avlfib,line width=0.8pt]
          (0,0.15) -- (0,-4.35);
        \node[color=avlfib,font=\small,anchor=east,rotate=90] at (-0.28,-2.1) {height 7};
      }
      \only<9->{
        \draw[decorate,decoration={brace,amplitude=5pt},avlpthree,line width=1pt]
          (0,0.15) -- (0,-3.77);
        \node[color=avlpthree,font=\small\bfseries,anchor=east,rotate=90]
          at (-0.28,-1.8) {height 6};
      }
    \end{scope}
  \end{tikzpicture}
  \end{center}

  \vspace{-0.5em}
  \begin{center}
    \only<1>{{\color{avlmuted}The smallest AVL tree of height 7.
              Every internal node is at $\pm1$ --- \fib{no spare room anywhere}.}}
    \only<2>{{\color{avlmuted}Delete \viol{one leaf}. Just one.}}
    \only<3>{{\color{avlmuted}Depth 3 is now unbalanced.}}
    \only<4>{{\color{avlmuted}\viol{Rotation 1} fires. The subtree gets shorter.}}
    \only<5>{{\color{avlmuted}That shortness breaks depth 2.}}
    \only<6>{{\color{avlmuted}\viol{Rotation 2} fires. Still shorter.}}
    \only<7>{{\color{avlmuted}That shortness breaks depth 1.}}
    \only<8>{{\color{avlmuted}\viol{Rotation 3} fires. It reaches the top.}}
    \only<9->{{\large\viol{\textbf{3 rotations}}\quad\quietc{$\cdot$}\quad
               height \textbf{7} $\rightarrow$ \color{avlpthree}\textbf{6}}\\[0.3em]
              {\color{avlmuted}One leaf removed. The whole tree got shorter.}}
  \end{center}

  \note{Step through this one click at a time. One leaf leaves. Circle the
  node that breaks first --- pause there, let the audience see WHY before
  anything moves. Then the arc fires and the rotation happens. Each fix makes
  that subtree shorter, which breaks the node above it, so the same two-beat
  pattern repeats at depth two, then depth one. The damage climbs all the way
  to the top and the whole tree drops from height seven to height six.}
\end{frame}

      \addplot[color=avlfib,thick,dashed] coordinates {(2,1)(14,1)};

  \end{tikzpicture}
  \end{center}

  \vspace{0.1em}
  \begin{center}
  \setlength{\tabcolsep}{13pt}
  \renewcommand{\arraystretch}{1.25}
  {\small
  \begin{tabular}{lcc}
    \color{avlmuted}one million nodes & \color{avlmuted}insertion & \color{avlmuted}deletion\\\hline
    worst case rotations & \fib{\textbf{1}} & \viol{\textbf{13}}\\
  \end{tabular}}
  \end{center}

  \vspace{0.3em}
  \begin{center}
    {\color{avlmuted}\small Measured by running a real AVL tree for every
     height from 2 to 14.}
  \end{center}

  \note{Insertion is flat at one, forever. Deletion grows with the height.
  At one million nodes the tree is 27 tall, so a single delete can cost 13
  rotations where an insert costs one.}
\end{frame}
