% ---------------------------------------------------------------------------
%  Rotation motion.
%
%  Drawn the way it is drawn on a board: a PIN goes into one node from below,
%  and the whole tree -- imagined as a stiff string of beads -- is TURNED about
%  that pin. One big arc sweeps over the top of the tree in the direction of
%  the turn. Nothing hangs from anything; nothing is lifted. The tree rotates.
%
%    right rotation -> clockwise      -> arc runs up the left, over, down right
%    left  rotation -> anticlockwise  -> arc runs up the right, over, down left
%
%  The pin marks the node the rotation is NAMED at -- the unbalanced one, the
%  one that comes down. A double rotation gets two pins and two arcs, in order.
%
%  Beamer overlays are jump cuts, so GHOSTING carries the eye across: the
%  post-rotation frame redraws the pre-rotation positions at low opacity, with
%  a faint thread to where each node ended up.
% ---------------------------------------------------------------------------

% Pin and arcs are drawn in the picture's own units. A dense tree needs a
% smaller pin: \setpinscale{0.5} before drawing one. 1 is the default.
\newcommand{\avlpinscale}{1}
\newcommand{\setpinscale}[1]{\renewcommand{\avlpinscale}{#1}}

% \pivotpin[<colour>]{<coord>}
% The pin stands UNDER the node, not in it: it is a fixed post in the ground
% and the node slides over it like every other node, so the tip stops short of
% the rim rather than touching it. Straight down is also the one direction that
% never collides with a child edge, and it is where the mark sits on the board.
\newcommand{\pivotpin}[2][avlfib]{%
  \begin{scope}[shift={(#2)},scale=\avlpinscale]
    \draw[draw=#1, line width=0.9pt, line cap=round] (0,-0.95) -- (0,-0.60);
    \fill[#1] (0,-0.99) circle (0.095);
    \fill[#1] (0,-0.58) circle (0.024);
  \end{scope}%
}

% \rotright[<colour>]{<pivot>}{<radius>}
% RIGHT rotation = turn the tree clockwise about the pin. The arc starts low on
% the left, sweeps over the top, and points down on the right.
\newcommand{\rotright}[3][avlfib]{%
  \draw[-{Stealth[length=5pt,width=4pt]}, draw=#1, line width=1.1pt]
    (#2) ++(205:#3) arc (205:-25:#3);%
}

% \rotleft[<colour>]{<pivot>}{<radius>} -- the mirror image, anticlockwise.
\newcommand{\rotleft}[3][avlfib]{%
  \draw[-{Stealth[length=5pt,width=4pt]}, draw=#1, line width=1.1pt]
    (#2) ++(-25:#3) arc (-25:205:#3);%
}

% \ghostnode{x}{y}{label} -- a faded afterimage of a node at its OLD spot.
% Dashed outline and NO fill on purpose: a solid disc reads as a live node, and
% a ghost sitting directly under a real one then looks like its child.
\newcommand{\ghostnode}[3]{%
  \node[circle, draw=avlmuted, dashed, line width=0.4pt, fill=none,
        text=avlmuted, opacity=0.55, inner sep=1pt, minimum size=6.5mm,
        font=\footnotesize] at (#1,#2) {#3};%
}
% \ghostpath{from}{to} -- faint thread from the old spot to the new one, with
% an arrowhead so the direction of travel is stated, not inferred.
\newcommand{\ghostpath}[2]{%
  \draw[-{Stealth[length=3.5pt]}, draw=avlmuted, opacity=0.55, dotted,
        line width=0.6pt] (#1) -- (#2);%
}
